\section{قاعده تفکیک}
هر ردیف در ماتریس تخصیص از پنج قاعده زیر بیرون آمده است. وقتی قابلیت جدیدی به محصول اضافه شد،
با همین قواعد جای آن را تعیین کنید تا تصمیم‌ها در طول زمان سازگار بمانند.
\begin{enumerate}
\item \textbf{می‌خواند، می‌سنجد، می‌خرد $\rightarrow$ وب.} صفحه بزرگ، صفحه‌کلید، اعتماد پرداخت، و مهم‌تر از همه ایندکس‌شدن در موتور جستجو.
\item \textbf{مصرف می‌کند، عادت روزانه است، در حرکت است $\rightarrow$ موبایل.} پخش پس‌زمینه، قفل صفحه، هدفون، بلوتوث خودرو، آفلاین، نوتیفیکیشن.
\item \textbf{فایل آپلود می‌کند یا ویرایش سنگین دارد $\rightarrow$ وب.} استودیو، پورتال ناشر، پنل سازمانی، پنل ادمین.
\item \textbf{به میکروفن، حسگر یا سخت‌افزار دستگاه نیاز دارد $\rightarrow$ موبایل.} ضبط صدا، تعامل صوتی، جستجوی صوتی، دستیار صوتی.
\item \textbf{کودک مصرف‌کننده است ولی والد تصمیم‌گیرنده $\rightarrow$ دو سطح.} کنترل و خرید در وب، مصرف در موبایل.
\end{enumerate}

\begin{notebox}{قاعده‌ای که کل معماری فرانت را تعیین می‌کند}
هیچ قابلیتی در سطح \lt{API} «فقط وب» یا «فقط موبایل» نیست. بک‌اند همه‌چیز را به هر سه سطح ارائه می‌دهد
و تفکیک تنها در لایه \textbf{رابط کاربری و اولویت تجربه} اعمال می‌شود. اگر این قاعده رعایت نشود،
با هر تغییر تصمیم محصولی باید بک‌اند بازنویسی شود.
\end{notebox}

\section{معماری سوپر‌اپ: یک اپ، چند سرویس}
\label{sec:superapp}
هدف این است که بزرگسال، کودک و خالق محتوا مانند اسنپ‌فود و اسنپ‌پی درون یک اپ جامع مدیریت شوند.
این جهت‌گیری درست است، اما باید مشخص شود کدام بخش از الگوی اسنپ به کتاپاد منتقل می‌شود و کدام بخش نه ---
چون کپی کامل آن الگو، چند سال کار مهندسی برای سه سرویس است.

الگوی اسنپ را حذف می‌کند و تقریباً تمام منفعتش را نگه می‌دارد.

اپ مستقل «کتاپاد کودک» صرفاً یک ورودی دیگر در \cd{apps/} است که فقط \cd{service\_kids} را ثبت می‌کند.
نتیجه: هم کودک داخل اپ جامع هست (برای خانواده‌ای که از قبل کاربر است)، هم به‌صورت اپ مستقل
در کافه‌بازار منتشر می‌شود (برای والدی که «قصه صوتی کودک» جستجو می‌کند) --- با یک کدبیس.
\end{kidbox}

\subsection{قرارداد ماژول سرویس}
هر سرویس یک کلاس ثابت پیاده می‌کند و پوسته چیزی درباره درون آن نمی‌داند.

\begin{english}
\begin{Verbatim}[fontsize=\footnotesize,xleftmargin=4mm]
abstract class ServiceModule {
  String get id;                      // "listen" | "kids" | "studio"
  String get titleKey;
  IconData get icon;
  ThemeData themeFor(Brightness b);   // per-service theme
  List<RouteBase> routes();           // mounted under /{id}/...
  bool isAvailable(SessionContext c); // roles, profile type, flags
  Future<void> warmUp();              // lazy - called on first entry
}
\end{Verbatim}
\end{english}

پوسته ماژول‌ها را در \cd{ServiceRegistry} ثبت می‌کند و مسیریاب (\lt{go\_router}) مسیرهای هر
ماژول را زیر پیشوند خودش سوار می‌کند: \cd{/listen/book/123}، \cd{/kids/story/45}،
\cd{/studio/upload}. \lt{deep-link} و \lt{universal link} به همین مسیرها نگاشت می‌شوند.

\textbf{متادیتا زودهنگام، کد دیرهنگام.} فقط شناسه و آیکون و شرط دسترسیِ ماژول‌ها در زمان
راه‌اندازی خوانده می‌شود؛ بارگذاری واقعی کد هر سرویس هنگام اولین ورود انجام می‌گیرد،
وگرنه زمان \lt{cold start} با هر سرویس جدید بدتر می‌شود.

\subsection{رجیستری سرویس سمت سرور}
این همان چیزی است که الگوی اسنپ را واقعاً ارزشمند می‌کند و اگر حذف شود، سوپر‌اپ فقط
یک اپ بزرگ‌تر است. پوسته در هر بار اجرا می‌پرسد «این کاربر چه سرویس‌هایی می‌بیند».

\begin{english}
\begin{Verbatim}[fontsize=\footnotesize,xleftmargin=4mm]
GET /api/v1/me/services

{
  "services": [
    { "id": "listen", "order": 1, "enabled": true,
      "badge": null,   "minAppVersion": "1.0.0" },
    { "id": "kids",   "order": 2, "enabled": true,
      "badge": "new",  "minAppVersion": "1.2.0" },
    { "id": "studio", "order": 3, "enabled": false,
      "reason": "creator_role_required",
      "cta": { "label": "apply", "route": "/studio/apply" } }
  ]
}
\end{Verbatim}
\end{english}

با این \lt{endpoint} می‌توانید سرویس تازه را برای ۵ درصد کاربران روشن کنید، سرویس معیوب را
بدون انتشار نسخه جدید خاموش کنید، و ترتیب کاشی‌ها را بر اساس رفتار کاربر تغییر دهید.
\cd{minAppVersion} اجازه می‌دهد سرویسی که به قابلیت جدید کلاینت نیاز دارد روی نسخه‌های قدیمی
پنهان بماند --- که در ایران اهمیت مضاعف دارد، چون کاربر کافه‌بازار دیر آپدیت می‌کند.

\subsection{ورود به سرویس: پروفایل، نه گرید}
اسنپ گرید سرویس می‌گذارد چون سرویس‌هایش \textbf{کار} هستند. سرویس‌های شما بیشتر
\textbf{آدم} هستند، پس الگوی درست ورودی، سوییچ پروفایل است:
\begin{itemize}
\item در اجرای اول پس از ورود: «چه کسی گوش می‌دهد؟» --- پروفایل بزرگسال یا پروفایل کودک.
\item پروفایل کودک $\rightarrow$ مستقیم وارد \cd{service\_kids} با تم و ناوبری خودش.
\item پروفایل بزرگسال $\rightarrow$ وارد \cd{service\_listen}.
\item \textbf{استودیو} چون یک نقش است نه یک شخص، به‌صورت کاشی در حساب کاربری ظاهر می‌شود و
فقط برای کاربری که نقش \cd{creator} دارد فعال است.
\end{itemize}

\begin{warnbox}{قاعده‌ای که اگر شکسته شود، صوت اپ خراب می‌شود}
\textbf{موتور صوت باید در پوسته زندگی کند، نه داخل سرویس‌ها.} روی هر دستگاه فقط
\textbf{یک} \lt{audio session} و یک نوتیفیکیشن پخش می‌تواند وجود داشته باشد. اگر
\cd{service\_listen} و \cd{service\_kids} هرکدام نمونه \cd{just\_audio} خودشان را بسازند،
نتیجه دو کنترل روی قفل صفحه، جنگ بر سر \lt{audio focus} و پخش هم‌زمان دو فایل است.
\cd{core\_audio} یک \lt{singleton} در پوسته باشد و سرویس‌ها فقط \textbf{صف پخش} را به آن بدهند.
\end{warnbox}

\begin{warnbox}{ایمنی کودک در پوسته اعمال می‌شود، نه در سرویس}
اگر حالت کودک صرفاً یک مسیر در مسیریاب باشد، یک \lt{deep-link} به \cd{/listen/...} آن را دور می‌زند.
پوسته باید \cd{SessionMode} نگه دارد؛ مسیریاب هر مسیر خارج از سرویس فعال را رد کند؛ و
\cd{core\_network} در همه درخواست‌ها هدر \cd{X-Profile-Id} بفرستد تا \textbf{بک‌اند نیز}
سیاست کودک را مستقل از کلاینت اعمال کند. اعمال یک‌طرفه در کلاینت کافی نیست.
\end{warnbox}

\subsection{استودیو به‌صورت \lt{WebView}}
استودیوی سازنده وب‌محور تعریف شده است. در اپ، آن را دوباره ننویسید:
\cd{service\_studio} یک \lt{WebView} سخت‌شده روی همان استودیوی وب باشد، به‌علاوه یک پل بومی
برای دو کار که وب نمی‌تواند: \textbf{ضبط با میکروفن} و \textbf{آپلود فایل حجیم در پس‌زمینه}.
این دقیقاً همان کاری است که سوپر‌اپ‌ها برای سرویس‌های کم‌تکرار انجام می‌دهند و ماه‌ها کار
صرفه‌جویی می‌کند. توکن با \lt{one-time code} از پوسته به \lt{WebView} پاس داده شود،
نه با اشتراک‌گذاری \lt{cookie}.

\section{چهار اهرم بهره‌وری}
\label{sec:leverage}
بدون این چهار مورد، نگهداری هم‌زمان چهار کدبیس برای یک تیم کوچک شدنی نیست.

\subsection{قرارداد \lt{API} به‌عنوان تنها منبع حقیقت}
یک فایل \cd{openapi.yaml} نوشته می‌شود و از آن سه خروجی تولید می‌گردد: واسط‌های سرور در \lt{Go}
با \lt{oapi-codegen}، کلاینت \lt{TypeScript} با \lt{orval}، و کلاینت \lt{Dart} با \lt{openapi-generator}.
وقتی یک فیلد تغییر می‌کند، هر سه کلاینت \textbf{در زمان کامپایل} خطا می‌دهند نه در زمان اجرا.

\subsection{توکن‌های طراحی از یک منبع}
یک فایل \lt{JSON} شامل رنگ، فاصله، شعاع و تایپوگرافی؛ از آن \cd{tailwind.config} و
\cd{ThemeData} فلاتر تولید می‌شود. بدون این، وب و اپ ظرف سه ماه واگرا می‌شوند.

\subsection{پنل‌ها ساخته نشوند، پیکربندی شوند}
ادمین، ناشر، سازمان و استودیو چهار پنل با نیازهای \lt{CRUD} مشابه‌اند. با \lt{Refine} روی همان
\lt{API}، این چهار پنل چند هفته کار است نه چند ماه.

\subsection{سرویس‌ها ماژول باشند نه کدبیس}
معماری بخش \ref{sec:superapp} اجازه می‌دهد از یک مجموعه بسته، هم اپ جامع و هم اپ مستقل کودک
ساخته شود --- دو خروجی، یک بار نگهداری.

\begin{warnbox}{قاعده‌ای که از سه‌برابر شدن کار جلوگیری می‌کند}
هیچ منطق کسب‌وکاری در فرانت نوشته نشود. محاسبه سطح گیمیفیکیشن، شرط تخفیف، سیاست کودک و
محاسبه \cd{Entitlement} همه در \lt{Go} انجام شوند و فرانت تنها نمایش دهد. اگر این قاعده شکسته شود،
هر قاعده \textbf{سه بار} نوشته می‌شود --- وب، موبایل، کودک --- و سپس سه نسخه از هم واگرا می‌شوند.
\end{warnbox}
